**tenant-validation-flow.json:**

Flow name: "ACV - Tenant Audit Validation (Daily)"
Description: "Runs tenant-level audit configuration validation daily at 6:00 AM UTC. Detects configuration drift and sends alerts via Teams and email."

Structure:
```
triggers:
  Recurrence:
    type: Recurrence
    recurrence:
      frequency: Day
      interval: 1
      schedule:
        hours: [6]
        minutes: [0]
      timeZone: "UTC"

actions:
  Scope_Try:
    type: Scope
    actions:
      Initialize_RunbookParams:
        type: InitializeVariable
        inputs:
          variables:
            - name: RunbookParams
              type: object
              value:
                Zone: "@{triggerOutputs()?['body']?['Zone'] ?? 'Zone3'}"  # Default Zone3 for enterprise
                DataverseUrl: "@{variables('DataverseUrl')}"
                TenantId: "@{variables('TenantId')}"
                ClientId: "@{variables('ClientId')}"
                CertificateThumbprint: "@{variables('CertificateThumbprint')}"

      Create_Automation_Job:
        type: ApiConnection
        inputs:
          host:
            connection:
              name: "@parameters('$connections')['azureautomation']['connectionId']"
            api:
              name: azureautomation
          method: post
          path: "/subscriptions/@{variables('SubscriptionId')}/resourceGroups/@{variables('ResourceGroup')}/providers/Microsoft.Automation/automationAccounts/@{variables('AutomationAccount')}/jobs"
          body:
            properties:
              runbook:
                name: Start-TenantValidationRunbook
              parameters: "@variables('RunbookParams')"
              runOn: ""

      Wait_For_Job:
        type: Until
        expression: "@or(equals(body('Get_Job_Status')?['properties']?['status'], 'Completed'), equals(body('Get_Job_Status')?['properties']?['status'], 'Failed'), equals(body('Get_Job_Status')?['properties']?['status'], 'Suspended'))"
        limit:
          count: 240
          timeout: PT2H
        actions:
          Delay_30s:
            type: Wait
            inputs:
              interval:
                count: 30
                unit: Second
          Get_Job_Status:
            type: ApiConnection  # Azure Automation Get Job

      Check_Job_Failed:
        type: If
        expression: "@equals(body('Get_Job_Status')?['properties']?['status'], 'Failed')"
        actions:
          # Yes branch: Send error email, terminate
          Send_Job_Error_Email: ...
        else:
          # No branch: Continue to get output

      Get_Job_Output:
        type: ApiConnection  # Azure Automation Get Job Output

      Parse_Results:
        type: ParseJson
        inputs:
          content: "@body('Get_Job_Output')"
          schema: { ... }  # Schema matching Start-TenantValidationRunbook output

      Check_Alert_Required:
        type: If
        expression: "@equals(body('Parse_Results')?['AlertRequired'], true)"
        actions:
          # Yes branch: Alert routing
          Check_Severity_For_Teams:
            type: If
            expression: "@or(equals(body('Parse_Results')?['AlertSeverity'], 'Failed'), equals(body('Parse_Results')?['AlertSeverity'], 'Error'))"
            actions:
              Post_Teams_Card:
                type: ApiConnection  # Teams: Post adaptive card in a chat or channel
                inputs:
                  body:
                    recipient: "@variables('TeamsChannelId')"
                    messageBody: <adaptive-card-tenant-alert.json with expressions>
            else: {}  # Warning severity: email only

          Send_Alert_Email:
            type: ApiConnection  # Office 365 Outlook: Send an email (V2)
            inputs:
              body:
                to: "@variables('ComplianceDistributionList')"
                subject: "[ALERT] Audit Configuration Drift - Tenant - @{body('Parse_Results')?['OverallStatus']}"
                body: <HTML email body with validation details>
                importance: "@if(or(equals(body('Parse_Results')?['AlertSeverity'], 'Failed'), equals(body('Parse_Results')?['AlertSeverity'], 'Error')), 'High', 'Normal')"

  Scope_Catch:
    type: Scope
    runAfter:
      Scope_Try:
        - Failed
        - TimedOut
    actions:
      Send_Error_Email:
        type: ApiConnection
        inputs:
          body:
            to: "@variables('ComplianceDistributionList')"
            subject: "[ERROR] Tenant Audit Validation Flow Failed"
            body: "The automated tenant audit validation flow encountered an error. Flow: @{workflow()['name']}, Run ID: @{workflow()['run']['name']}, Timestamp: @{utcNow()}"
            importance: High
```

Include all necessary variable initializations at the top:
- DataverseUrl, TenantId, ClientId, CertificateThumbprint (from environment variables or connection reference)
- SubscriptionId, ResourceGroup, AutomationAccount (Azure Automation location)
- TeamsChannelId (target Teams channel for adaptive card alerts)
- ComplianceDistributionList (email distribution list)
